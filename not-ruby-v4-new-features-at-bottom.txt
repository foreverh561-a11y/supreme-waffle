local function makeTab(name,text)
	local button = Instance.new("TextButton")
	button.Name = name.."Tab"
	button.Size = UDim2.new(1,0,0,42)
	button.Text = text
	button.TextSize = 10
	button.Font = Enum.Font.GothamBlack
	button.TextColor3 = TEXT
	button.BackgroundColor3 = SURFACE
	button.AutoButtonColor = false
	button.Parent = sidebar
	corner(button,10)

	local scale = Instance.new("UIScale")
	scale.Scale = 1
	scale.Parent = button

	button.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Touch
			or input.UserInputType == Enum.UserInputType.MouseButton1 then
			TweenService:Create(
				scale,
				TweenInfo.new(.08,Enum.EasingStyle.Quad,Enum.EasingDirection.Out),
				{Scale = .96}
			):Play()
		end
	end)

	button.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Touch
			or input.UserInputType == Enum.UserInputType.MouseButton1 then
			TweenService:Create(
				scale,
				TweenInfo.new(.12,Enum.EasingStyle.Quint,Enum.EasingDirection.Out),
				{Scale = 1}
			):Play()
		end
	end)

	tabButtons[name] = button
	return button
end
local function applyTabVisuals()
	for name,button in pairs(tabButtons) do
		local selected = name == currentTab
		TweenService:Create(
			button,
			TweenInfo.new(.14,Enum.EasingStyle.Quint,Enum.EasingDirection.Out),
			{
				BackgroundColor3 = selected and accent or SURFACE,
				TextColor3 = selected and ACTIVE_TEXT or TEXT
			}
		):Play()
	end

	for name,page in pairs(pages) do
		page.ScrollBarImageColor3 = accent

		if name == currentTab then
			page.Visible = true
			page.Position = UDim2.fromOffset(10,0)
			page.ScrollBarImageTransparency = .55

			TweenService:Create(
				page,
				TweenInfo.new(.18,Enum.EasingStyle.Quint,Enum.EasingDirection.Out),
				{
					Position = UDim2.fromOffset(0,0),
					ScrollBarImageTransparency = 0
				}
			):Play()
		else
			page.Visible = false
		end
	end
end

	if not pages[name] then

end

	local lastSampleTime = nil
	local airborneTime = 0
	local signalCounts = {SPEED = 0,TELEPORT = 0,AIR = 0,HOVER = 0,UPWARD = 0,FLING = 0,FORCE = 0,JUMP = 0,NOCLIP = 0}
	local executorAliases = {
		delta = "DELTA",
		wave = "WAVE",
		arceus = "ARCEUS X",
		["arceus x"] = "ARCEUS X",
		codex = "CODEX",
		fluxus = "FLUXUS",
		hydrogen = "HYDROGEN"
	}

	local function normalizeExecutorName(value)
		if typeof(value) ~= "string" then
			return nil
		end
		local cleaned = string.gsub(value,"^%s*(.-)%s*$","%1")
		if cleaned == "" then
			return nil
		end
		return executorAliases[string.lower(cleaned)] or string.upper(cleaned)
	end

	local function resetInspectorHistory()
		lastPosition = nil
		lastSampleTime = nil
		airborneTime = 0
		signalCounts.SPEED = 0
		signalCounts.TELEPORT = 0
		signalCounts.AIR = 0
		signalCounts.HOVER = 0
		signalCounts.UPWARD = 0
		signalCounts.FLING = 0
		signalCounts.FORCE = 0
		signalCounts.JUMP = 0
		signalCounts.NOCLIP = 0
	end

	local function getGameWalkSpeedBaseline()
		local speeds = {}

		for _,otherPlayer in ipairs(Players:GetPlayers()) do
			local otherCharacter = otherPlayer.Character
			local otherHumanoid =
				otherCharacter
				and otherCharacter:FindFirstChildOfClass("Humanoid")

			if otherHumanoid and otherHumanoid.Health > 0 then
				table.insert(speeds,otherHumanoid.WalkSpeed)
			end
		end

		if #speeds == 0 then
			return 16
		end

		table.sort(speeds)

		-- Use the upper-middle server speed instead of assuming Roblox's
		-- default 16. If the game intentionally makes everybody faster,
		-- that becomes the detector's normal baseline.
		local index = math.clamp(
			math.ceil(#speeds * .75),
			1,
			#speeds
		)

		return math.max(16,speeds[index] or 16)
	end

	local function getGameJumpBaseline()
		local values = {}

		for _,otherPlayer in ipairs(Players:GetPlayers()) do
			local otherCharacter = otherPlayer.Character
			local otherHumanoid =
				otherCharacter
				and otherCharacter:FindFirstChildOfClass("Humanoid")

			if otherHumanoid and otherHumanoid.Health > 0 then
				local value =
					otherHumanoid.UseJumpPower
					and otherHumanoid.JumpPower
					or otherHumanoid.JumpHeight
				table.insert(values,value)
			end
		end

		if #values == 0 then
			return 50
		end

		table.sort(values)
		local index =
			math.clamp(math.ceil(#values*.75),1,#values)

		return values[index] or 50
	end

	local function hasCharacterForceObject(selectedCharacter)
		for _,object in ipairs(selectedCharacter:GetDescendants()) do
			if object:IsA("BodyVelocity")
				or object:IsA("BodyAngularVelocity")
				or object:IsA("BodyPosition")
				or object:IsA("BodyGyro")
				or object:IsA("LinearVelocity")
				or object:IsA("AngularVelocity")
				or object:IsA("VectorForce")
				or object:IsA("AlignPosition")
				or object:IsA("AlignOrientation") then
				return true,object.ClassName
			end
		end

		return false,nil
	end

	local function hasGroundBelow(selectedCharacter,selectedRoot,distance)
		local params = RaycastParams.new()
		params.FilterType = Enum.RaycastFilterType.Exclude
		params.FilterDescendantsInstances = {selectedCharacter}

		local result = workspace:Raycast(
			selectedRoot.Position,
			Vector3.new(0,-math.max(4,distance or 12),0),
			params
		)

		return result ~= nil
	end

	local function getSelectedParts()
		if not selectedPlayer or not selectedPlayer.Parent then
			return nil,nil,nil
		end
		local selectedCharacter = selectedPlayer.Character
		if not selectedCharacter then
			return nil,nil,nil
		end
		local selectedHumanoid = selectedCharacter:FindFirstChildOfClass("Humanoid")
		local selectedRoot =
			selectedCharacter:FindFirstChild("HumanoidRootPart")
			or selectedCharacter.PrimaryPart
		return selectedCharacter,selectedHumanoid,selectedRoot
	end

	local function updateInspection()
		if not selectedPlayer or not selectedPlayer.Parent then
			selectedTitle.Text = "NO PLAYER SELECTED"
			selectedUser.Text = "--"
			inspectAvatar.Image = ""
			verdictLabel.Text = "STATUS: --"
			verdictLabel.TextColor3 = MUTED
			executorLabel.Text = "EXECUTOR: --"
			executorLabel.TextColor3 = TEXT
			signalsLabel.Text = "SIGNALS: --"
			return
		end

		selectedTitle.Text = selectedPlayer.DisplayName
		selectedUser.Text = "@"..selectedPlayer.Name.."  •  "..tostring(selectedPlayer.UserId)

		local selectedCharacter,selectedHumanoid,selectedRoot = getSelectedParts()
		if not selectedCharacter or not selectedRoot then
			verdictLabel.Text = "STATUS: CHARACTER UNAVAILABLE"
			verdictLabel.TextColor3 = MUTED
			executorLabel.Text = "EXECUTOR: --"
			signalsLabel.Text = "SIGNALS: waiting for character"
			return
		end

		local now = os.clock()
		if lastPosition and lastSampleTime then
			local dt = math.max(now - lastSampleTime,.001)
			local delta = selectedRoot.Position - lastPosition
			local horizontal =
				Vector3.new(delta.X,0,delta.Z).Magnitude / dt

			local gameWalkSpeed = getGameWalkSpeedBaseline()
			local playerWalkSpeed =
				selectedHumanoid
				and math.max(selectedHumanoid.WalkSpeed,0)
				or gameWalkSpeed

			-- GAME-SPEED-AWARE CHECK:
			-- Fast movement by itself is not enough anymore. The selected
			-- player's WalkSpeed must ALSO be a clear outlier compared with
			-- the rest of the server. This stops games with naturally high
			-- movement speed from being called suspicious.
			local walkSpeedOutlier =
				playerWalkSpeed > (
					strictBaseplateMode
					and math.max(
						gameWalkSpeed * 1.35,
						gameWalkSpeed + 8
					)
					or math.max(
						gameWalkSpeed * 1.65,
						gameWalkSpeed + 20
					)
				)

			local movementTooFast =
				horizontal > (
					strictBaseplateMode
					and math.max(
						gameWalkSpeed * 2.35,
						playerWalkSpeed * 1.7,
						52
					)
					or math.max(
						gameWalkSpeed * 3.5,
						playerWalkSpeed * 2.25,
						110
					)
				)

			if walkSpeedOutlier and movementTooFast then
				signalCounts.SPEED += 1
			else
				-- Old spikes slowly clear instead of permanently marking
				-- somebody suspicious after one lag/game-speed burst.
				signalCounts.SPEED =
					math.max(0,signalCounts.SPEED - 1)
			end

			local teleportDistance =
				strictBaseplateMode and 75 or 180

			if delta.Magnitude > teleportDistance
				and dt < .65 then
				signalCounts.TELEPORT += 1
			end
			local velocity = selectedRoot.AssemblyLinearVelocity
			local angularVelocity =
				selectedRoot.AssemblyAngularVelocity.Magnitude

			-- Very high spin/velocity is common with fling/physics scripts.
			local flingLinearThreshold =
				strictBaseplateMode and 95 or 165
			local flingAngularThreshold =
				strictBaseplateMode and 18 or 42

			if velocity.Magnitude >= flingLinearThreshold
				or angularVelocity >= flingAngularThreshold then
				signalCounts.FLING += 1
			else
				signalCounts.FLING =
					math.max(0,signalCounts.FLING - 1)
			end

			-- In a simple/baseplate game, character force objects are a useful
			-- extra clue. They are NOT treated as proof by themselves outside
			-- strict mode because legitimate games can also use constraints.
			local hasForceObject =
				hasCharacterForceObject(selectedCharacter)

			if hasForceObject then
				signalCounts.FORCE +=
					strictBaseplateMode and 1 or .25
			else
				signalCounts.FORCE =
					math.max(0,signalCounts.FORCE - .5)
			end

			if selectedHumanoid then
				local gameJump = getGameJumpBaseline()
				local selectedJump =
					selectedHumanoid.UseJumpPower
					and selectedHumanoid.JumpPower
					or selectedHumanoid.JumpHeight

				local jumpLimit =
					strictBaseplateMode
					and math.max(gameJump*1.35,gameJump+10)
					or math.max(gameJump*1.8,gameJump+30)

				if selectedJump > jumpLimit then
					signalCounts.JUMP += 1
				else
					signalCounts.JUMP =
						math.max(0,signalCounts.JUMP - 1)
				end
			end

			-- Strict mode checks whether the straight replicated movement path
			-- crossed a collidable object. Repeated hits can indicate noclip.
			if strictBaseplateMode and delta.Magnitude >= 4 then
				local pathParams = RaycastParams.new()
				pathParams.FilterType =
					Enum.RaycastFilterType.Exclude
				pathParams.FilterDescendantsInstances =
					{selectedCharacter}

				local pathHit = workspace:Raycast(
					lastPosition,
					delta,
					pathParams
				)

				if pathHit
					and pathHit.Instance
					and pathHit.Instance:IsA("BasePart")
					and pathHit.Instance.CanCollide then
					signalCounts.NOCLIP += 1
				else
					signalCounts.NOCLIP =
						math.max(0,signalCounts.NOCLIP - 1)
				end
			end

			local noGround =
				not hasGroundBelow(
					selectedCharacter,
					selectedRoot,
					12
				)

			local humanoidAir =
				selectedHumanoid
				and selectedHumanoid.FloorMaterial == Enum.Material.Air

			if humanoidAir and noGround then
				airborneTime += dt

				-- Hovering: clearly off the ground but barely falling/rising.
				if airborneTime >= (strictBaseplateMode and .75 or 1.35)
					and math.abs(velocity.Y) <= 5 then
					signalCounts.HOVER += 1
					airborneTime = .75
				end

				-- Controlled air travel: moving around while remaining far
				-- above any floor for long enough to rule out a normal jump.
				local horizontalAir =
					Vector3.new(
						velocity.X,
						0,
						velocity.Z
					).Magnitude

				if airborneTime >= (strictBaseplateMode and 1.25 or 2.25)
					and horizontalAir >= 5 then
					signalCounts.AIR += 1
					airborneTime = 1
				end

				-- Repeated/controlled upward travel while there is no nearby
				-- floor. A normal jump may spike once, so require repeats.
				if velocity.Y >= (strictBaseplateMode and 12 or 18) then
					signalCounts.UPWARD += 1
				else
					signalCounts.UPWARD =
						math.max(0,signalCounts.UPWARD - 1)
				end
			else
				airborneTime = 0
				signalCounts.UPWARD =
					math.max(0,signalCounts.UPWARD - 1)
			end
		end

		lastPosition = selectedRoot.Position
		lastSampleTime = now

		local signals = {}
		if signalCounts.SPEED >= 2 then
			table.insert(signals,"abnormal speed")
		end
		if signalCounts.TELEPORT >= 2 then
			table.insert(signals,"repeated position jumps")
		end
		if signalCounts.AIR >= 1 then
			table.insert(signals,"controlled airborne travel")
		end
		if signalCounts.HOVER >= 1 then
			table.insert(signals,"hovering with no ground below")
		end
		if signalCounts.UPWARD >= 3 then
			table.insert(signals,"repeated upward flight-like movement")
		end
		if signalCounts.FLING >= 2 then
			table.insert(signals,"extreme physics / fling-like movement")
		end
		if signalCounts.FORCE >= (strictBaseplateMode and 2 or 5) then
			table.insert(signals,"character force / mover object")
		end
		if signalCounts.JUMP >= 2 then
			table.insert(signals,"abnormal jump setting")
		end
		if strictBaseplateMode and signalCounts.NOCLIP >= 2 then
			table.insert(signals,"possible noclip through solid parts")
		end

		local serverFlag =
			selectedPlayer:GetAttribute("ExploitDetected") == true
		local serverReason =
			selectedPlayer:GetAttribute("ExploitReason")
		local executorName = normalizeExecutorName(
			selectedPlayer:GetAttribute("ExecutorName")
		)

		local speedOnlyServerFlag = false
		if serverFlag
			and typeof(serverReason) == "string"
			and serverReason ~= "" then
			local loweredReason = string.lower(serverReason)
			speedOnlyServerFlag =
				string.find(loweredReason,"speed",1,true) ~= nil
				and string.find(loweredReason,"flight",1,true) == nil
				and string.find(loweredReason,"air",1,true) == nil
				and string.find(loweredReason,"teleport",1,true) == nil
		end

		if serverFlag and not speedOnlyServerFlag then
			table.insert(signals,"server anti-cheat flag")
		elseif speedOnlyServerFlag then
			table.insert(signals,"fast movement (may be game speed)")
		end

		if typeof(serverReason) == "string"
			and serverReason ~= ""
			and not speedOnlyServerFlag then
			table.insert(signals,"server: "..serverReason)
		end

		local strongLocalSignal =
			signalCounts.TELEPORT >= 2
			or signalCounts.AIR >= 1
			or signalCounts.HOVER >= 1
			or signalCounts.UPWARD >= 3
			or signalCounts.SPEED >= 2
			or signalCounts.FLING >= 2
			or signalCounts.JUMP >= 2
			or (
				strictBaseplateMode
				and (
					signalCounts.FORCE >= 2
					or signalCounts.NOCLIP >= 2
				)
			)

		local suspicious =
			(serverFlag and not speedOnlyServerFlag)
			or executorName ~= nil
			or strongLocalSignal

		if suspicious then
			verdictLabel.Text = "STATUS: SUSPICIOUS"
			verdictLabel.TextColor3 = Color3.fromRGB(255,92,108)
		else
			verdictLabel.Text = "STATUS: NONE DETECTED"
			verdictLabel.TextColor3 = Color3.fromRGB(90,230,135)
		end

		if executorName then
			executorLabel.Text = "EXECUTOR: "..executorName
			executorLabel.TextColor3 = Color3.fromRGB(255,92,108)
		elseif suspicious then
			executorLabel.Text = "EXECUTOR: UNKNOWN"
			executorLabel.TextColor3 = Color3.fromRGB(255,170,95)
		else
			executorLabel.Text = "EXECUTOR: NONE"
			executorLabel.TextColor3 = Color3.fromRGB(90,230,135)
		end

		if #signals == 0 then
			signalsLabel.Text =
				"SIGNALS: none observed. This detects replicated behavior/character anomalies, not hidden script code. Exact executor names still require server-reported ExecutorName."
		else
			signalsLabel.Text = "SIGNALS: "..table.concat(signals,", ")
		end
	end

	local function selectInspectorPlayer(otherPlayer)
		if not otherPlayer or otherPlayer == player then
			return
		end

		selectedPlayer = otherPlayer
		resetInspectorHistory()

		task.spawn(function()
			local ok,image = pcall(function()
				return Players:GetUserThumbnailAsync(
					otherPlayer.UserId,
					Enum.ThumbnailType.HeadShot,
					Enum.ThumbnailSize.Size150x150
				)
			end)
			if ok and image and inspectAvatar.Parent and selectedPlayer == otherPlayer then
				inspectAvatar.Image = image
			end
		end)

		updateInspection()
		setStatus("INSPECTING @"..otherPlayer.Name)
	end

	selectPlayerButton.Activated:Connect(function()
		showPlayerPicker("SELECT PLAYER TO INSPECT",selectInspectorPlayer,false)
	end)

	refreshInspectButton.Activated:Connect(function()
		resetInspectorHistory()
		updateInspection()
		setStatus(selectedPlayer and "INSPECTION REFRESHED" or "SELECT A PLAYER FIRST")
	end)

	clearInspectButton.Activated:Connect(function()
		selectedPlayer = nil
		resetInspectorHistory()
		updateInspection()
		setStatus("INSPECTION CLEARED")
	end)

	Players.PlayerRemoving:Connect(function(leavingPlayer)
		if selectedPlayer == leavingPlayer then
			selectedPlayer = nil
			resetInspectorHistory()
			updateInspection()
		end
	end)

	task.spawn(function()
		while not menuDeleted do
			task.wait(.25)
			if selectedPlayer and currentTab == "EXPLOITER" and exploiterPage.Parent then
				updateInspection()
			end
		end
	end)
end

local function setToggle(button,on,label)
	if not button or not button.Parent then
		return
	end
	activeToggleButtons[button] = {
		on = on,
		label = label
	}
	if on then
		button.BackgroundColor3 = accent
		button.TextColor3 = Color3.fromRGB(255,255,255)
		button.Text = label.."  [ON]"
	else
		button.BackgroundColor3 = SURFACE_2
		button.TextColor3 = TEXT
		button.Text = label.."  [OFF]"
	end
end
local function refreshToggleColors()
	for button,data in pairs(activeToggleButtons) do
		if button and button.Parent then
			setToggle(button,data.on,data.label)
		end
	end
end
local function cycleTheme()
	themeIndex += 1
	if themeIndex > #themes then
		themeIndex = 1
	end

	accent = themes[themeIndex].color
	status.TextColor3 = accent
	sidebar.ScrollBarImageColor3 = accent

	applyTabVisuals()
	refreshToggleColors()

	-- MATCH makes every outline follow the selected theme.
	if gui:GetAttribute("OutlineMode") == "MATCH" then
		gui:SetAttribute("OutlineColor",accent)
		for _,object in ipairs(gui:GetDescendants()) do
			if object:IsA("UIStroke") then
				object.Color = accent
			end
		end
	end

	setStatus("THEME: "..themes[themeIndex].name)
end

local function applyMovementStats()
	if not humanoid then
		return
	end

RunService.RenderStepped:Connect(function()
	if menuDeleted then
		return
	end
	if not flying
		or not root
		or not root.Parent
		or not humanoid
		or not bodyGyro
		or not flightVelocity then
		return
	end
	camera = workspace.CurrentCamera
	if not camera then
		return
	end
	local direction = Vector3.zero
	local keyboardUsed = false
	local look = camera.CFrame.LookVector
	local right = camera.CFrame.RightVector
	if not pitchFlight then
		look = Vector3.new(look.X,0,look.Z)
		right = Vector3.new(right.X,0,right.Z)
		if look.Magnitude > .001 then
			look = look.Unit
		end
		if right.Magnitude > .001 then
			right = right.Unit
		end
	end
	if UIS:IsKeyDown(Enum.KeyCode.W) then
		direction += look
		keyboardUsed = true
	end
	if UIS:IsKeyDown(Enum.KeyCode.S) then
		direction -= look
		keyboardUsed = true
	end
	if UIS:IsKeyDown(Enum.KeyCode.D) then
		direction += right
		keyboardUsed = true
	end
	if UIS:IsKeyDown(Enum.KeyCode.A) then
		direction -= right
		keyboardUsed = true
	end
	if not keyboardUsed and humanoid.MoveDirection.Magnitude > .03 then
		local flatLook = Vector3.new(
			camera.CFrame.LookVector.X,
			0,
			camera.CFrame.LookVector.Z
		)
		local flatRight = Vector3.new(
			camera.CFrame.RightVector.X,
			0,
			camera.CFrame.RightVector.Z
		)
		if flatLook.Magnitude > .001 then
			flatLook = flatLook.Unit
		end
		if flatRight.Magnitude > .001 then
			flatRight = flatRight.Unit
		end
		local move = humanoid.MoveDirection
		local forwardAmount = move:Dot(flatLook)
		local rightAmount = move:Dot(flatRight)
		local mobileLook =
			pitchFlight and camera.CFrame.LookVector or flatLook
		local mobileRight =
			pitchFlight and camera.CFrame.RightVector or flatRight
		direction += mobileLook * forwardAmount
		direction += mobileRight * rightAmount
	end
	local vertical = 0
	if flyUp then
		vertical += 1
	end
	if flyDown then
		vertical -= 1
	end
	if UIS:IsKeyDown(Enum.KeyCode.Space)
		or UIS:IsKeyDown(Enum.KeyCode.E) then
		vertical += 1
	end
	if UIS:IsKeyDown(Enum.KeyCode.LeftControl)
		or UIS:IsKeyDown(Enum.KeyCode.Q) then
		vertical -= 1
	end
	direction += Vector3.new(0,vertical,0)
	if direction.Magnitude > 1 then
		direction = direction.Unit
	end
	flightVelocity.Velocity = direction * flySpeed
	local facing = camera.CFrame.LookVector
	local flatFacing = Vector3.new(facing.X,0,facing.Z)
	if flatFacing.Magnitude > .01 then
		bodyGyro.CFrame =
			CFrame.lookAt(
				root.Position,
				root.Position + flatFacing.Unit
			)
	end
end)

-- WALL / COLLISION PARTS
-- Shows nearby collidable geometry with red outlines. This is useful for
-- seeing actual solid surfaces instead of decorative/non-collidable parts.
gui:SetAttribute("WallPartsVisual",false)

local function setupWallPartsVisualizer()
	local scanRadius = 120
	local refreshDelay = .45
	local visualFolder = Instance.new("Folder")
	visualFolder.Name = "NotRubyWallPartVisuals"
	visualFolder.Parent = workspace
	local activeBoxes = {}
	local token = 0

	local function isCharacterPart(part)
		local model = part:FindFirstAncestorOfClass("Model")
		return model and model:FindFirstChildOfClass("Humanoid") ~= nil
	end

	local function clearBoxes()
		for part,box in pairs(activeBoxes) do
			if box then
				box:Destroy()
			end
			activeBoxes[part] = nil
		end
	end

	local function removeMissingParts(seen)
		for part,box in pairs(activeBoxes) do
			if not seen[part] or not part.Parent then
				if box then
					box:Destroy()
				end
				activeBoxes[part] = nil
			end
		end
	end

	local function getPartVisualColor(part)
		local smallest =
			math.min(part.Size.X,part.Size.Y,part.Size.Z)

		if smallest <= 1.15 then
			return Color3.fromRGB(255,205,55)
		end

		if part.CanCollide then
			return Color3.fromRGB(255,45,60)
		end

		return Color3.fromRGB(155,160,175)
	end

	local function addPart(part)
		if activeBoxes[part]
			or not part:IsA("BasePart")
			or part.Transparency >= 1
			or isCharacterPart(part) then
			return
		end

		local color = getPartVisualColor(part)
		local box = Instance.new("SelectionBox")
		box.Name = "NotRubyCollisionOutline"
		box.Adornee = part
		box.Color3 = color
		box.SurfaceColor3 = color
		box.SurfaceTransparency = .93
		box.LineThickness = .035
		box.Parent = visualFolder
		activeBoxes[part] = box
	end

	local function refresh()
		local currentRoot =
			player.Character
			and player.Character:FindFirstChild("HumanoidRootPart")

		if not currentRoot then
			clearBoxes()
			return
		end

		local overlap = OverlapParams.new()
		overlap.FilterType = Enum.RaycastFilterType.Exclude
		overlap.FilterDescendantsInstances = {
			player.Character,
			visualFolder
		}

		local seen = {}
		local nearby = workspace:GetPartBoundsInRadius(
			currentRoot.Position,
			scanRadius,
			overlap
		)

		for _,part in ipairs(nearby) do
			if part:IsA("BasePart")
				and part.Transparency < 1
				and not isCharacterPart(part) then
				seen[part] = true

				local existing = activeBoxes[part]
				if existing then
					local color = getPartVisualColor(part)
					existing.Color3 = color
					existing.SurfaceColor3 = color
				else
					addPart(part)
				end
			end
		end

		removeMissingParts(seen)
	end

	local function stop()
		token += 1
		clearBoxes()
		setToggle(
			wallPartsButton,
			false,
			"WALL / COLLISION PARTS"
		)
	end

	local function start()
		token += 1
		local myToken = token
		setToggle(
			wallPartsButton,
			true,
			"WALL / COLLISION PARTS"
		)

		task.spawn(function()
			while not menuDeleted
				and gui:GetAttribute("WallPartsVisual") == true
				and myToken == token do
				refresh()
				task.wait(refreshDelay)
			end

			if myToken == token then
				clearBoxes()
			end
		end)
	end

	wallPartsButton.Activated:Connect(function()
		gui:SetAttribute(
			"WallPartsVisual",
			not (gui:GetAttribute("WallPartsVisual") == true)
		)
	end)

	gui:GetAttributeChangedSignal("WallPartsVisual"):Connect(function()
		if gui:GetAttribute("WallPartsVisual") == true then
			start()
			setStatus("WALL PARTS // RED SOLID // GRAY NON-COLLIDE // YELLOW THIN")
		else
			stop()
			setStatus("WALL / COLLISION PARTS OFF")
		end
	end)

	gui.Destroying:Connect(function()
		clearBoxes()
		if visualFolder then
			visualFolder:Destroy()
		end
	end)
end

local function getRoot(model)
	if not model or not model:IsA("Model") then
		return nil
	end
	local preferred = {
		"HumanoidRootPart",
		"UpperTorso",
		"Torso",
		"Head"
	}
	for _,name in ipairs(preferred) do
		local part = model:FindFirstChild(name)
		if part and part:IsA("BasePart") then
			return part
		end
	end
	for _,object in ipairs(model:GetChildren()) do
		if object:IsA("BasePart") then
			return object
		end
	end
	return nil
end

local function refreshPlayers()
	clearMarkerTable(playerMarkers)
	if not playerLocator then
		return
	end

	local selectedUserId = tonumber(gui:GetAttribute("TrackerTargetUserId")) or 0
	local selectedName = tostring(gui:GetAttribute("TrackerTargetName") or "")
	local foundTarget = false
	local markedCount = 0

	for _,otherPlayer in ipairs(Players:GetPlayers()) do
		if otherPlayer ~= player
			and otherPlayer.Character
			and (
				selectedUserId == 0
				or otherPlayer.UserId == selectedUserId
			) then

			createMarker(
				otherPlayer.Character,
				otherPlayer.DisplayName.."  @"..otherPlayer.Name,
				false
			)

			markedCount += 1

			if selectedUserId ~= 0 and otherPlayer.UserId == selectedUserId then
				foundTarget = true
			end
		end
	end

	if trackerInfo and trackerInfo.Parent then
		local labels = {}
		for _,child in ipairs(trackerInfo:GetChildren()) do
			if child:IsA("TextLabel") then
				table.insert(labels,child)
			end
		end

		if #labels >= 2 then
			if selectedUserId == 0 then
				labels[2].Text =
					"Tracking all players  |  Marked: "..tostring(markedCount)
			elseif foundTarget then
				labels[2].Text =
					"Tracking only @"..selectedName
			else
				labels[2].Text =
					"Selected player is not currently available"
			end
		end
	end
end

local monsterWords = {
	"locust","monster","zombie","creature","beast","enemy",
	"alien","mutant","boss","npc","insect","bug","spider",
	"scorpion","demon","ghoul","crawler","creeper","behemoth",

local function normalizeName(name)
	return string.lower(
		string.gsub(name or "","[%s_%-%p]+"," ")
	)
end
local function containsMonsterWord(name)
	name = normalizeName(name)
	for _,word in ipairs(monsterWords) do
		if string.find(name,word,1,true) then
			return true
		end
	end
	return false
end
local function isPlayerCharacter(model)
	for _,otherPlayer in ipairs(Players:GetPlayers()) do
		if otherPlayer.Character == model then
			return true
		end
	end
	return false
end
local function isMonster(model)
	if not model or not model:IsA("Model") then
		return false
	end
	if isPlayerCharacter(model) then
		return false
	end
	if model:GetAttribute("IsMonster") == true
		or model:GetAttribute("Monster") == true
		or model:GetAttribute("IsEnemy") == true
		or model:GetAttribute("Enemy") == true then
		return true
	end
	if CollectionService:HasTag(model,"Monster")
		or CollectionService:HasTag(model,"Enemy")
		or CollectionService:HasTag(model,"Locust") then
		return true
	end
	return containsMonsterWord(model.Name)
end
local function findMonsterModel(object)
	local current = object
	while current and current ~= workspace do
		if current:IsA("Model") and isMonster(current) then
			return current
		end
		current = current.Parent
	end
	return nil
end
local function refreshMonsters()
	clearMarkerTable(monsterMarkers)
	if not monsterLocator then
		return
	end
	local found = {}
	local count = 0
	for _,object in ipairs(workspace:GetDescendants()) do
		if object:IsA("Model") then
			local monster = findMonsterModel(object)
			if monster
				and not found[monster]
				and not isPlayerCharacter(monster) then
				local monsterRoot = getRoot(monster)
				if monsterRoot
					and (
						not root
						or (root.Position-monsterRoot.Position).Magnitude <= scanRadius
					) then
					found[monster] = true
					createMarker(monster,monster.Name,true)
					count += 1
				end
			end
		end
	end
	setStatus("MONSTERS FOUND: "..tostring(count))
end

local function setupMiniTabDrag()
	local miniDragging = false
	local miniDragInput
	local miniDragStart
	local miniStartAbsolute
	local miniStartUDimPosition
	local miniWasDragged = false

	openButton.InputBegan:Connect(function(input)
		if input.UserInputType ~= Enum.UserInputType.Touch
			and input.UserInputType ~= Enum.UserInputType.MouseButton1 then
			return
		end

		-- Do not start dragging from the resize grip.
		local grip = openButton:FindFirstChild("MiniResizeGrip")
		if grip then
			local p = input.Position
			local gp = grip.AbsolutePosition
			local gs = grip.AbsoluteSize
			if p.X >= gp.X and p.X <= gp.X + gs.X
				and p.Y >= gp.Y and p.Y <= gp.Y + gs.Y then
				return
			end
		end

		miniDragging = true
		miniDragInput = input
		miniDragStart = input.Position
		miniStartAbsolute = openButton.AbsolutePosition
		miniStartUDimPosition = openButton.Position
		miniWasDragged = false
	end)

	UIS.InputChanged:Connect(function(input)
		if not miniDragging
			or not miniDragStart
			or not miniStartAbsolute
			or not miniStartUDimPosition then
			return
		end

		local validInput =
			input.UserInputType == Enum.UserInputType.MouseMovement
			or (
				input.UserInputType == Enum.UserInputType.Touch
				and input == miniDragInput
			)

		if not validInput then
			return
		end

		local delta = input.Position - miniDragStart

		if delta.Magnitude > 6 then
			miniWasDragged = true
		end

		camera = workspace.CurrentCamera
		local viewport =
			camera and camera.ViewportSize
			or Vector2.new(1000,700)

		local size = openButton.AbsoluteSize

		-- Preserve the original UDim2 and only apply the movement delta.
		-- This fixes the mobile jump-up bug caused by rebuilding Position
		-- from AbsolutePosition.
		local minDX = 4 - miniStartAbsolute.X
		local maxDX =
			viewport.X - miniStartAbsolute.X - size.X - 4
		local minDY = 4 - miniStartAbsolute.Y
		local maxDY =
			viewport.Y - miniStartAbsolute.Y - size.Y - 4

		local appliedX = math.clamp(
			delta.X,
			minDX,
			math.max(minDX,maxDX)
		)
		local appliedY = math.clamp(
			delta.Y,
			minDY,
			math.max(minDY,maxDY)
		)

		openButton.Position = UDim2.new(
			miniStartUDimPosition.X.Scale,
			miniStartUDimPosition.X.Offset + appliedX,
			miniStartUDimPosition.Y.Scale,
			miniStartUDimPosition.Y.Offset + appliedY
		)
	end)

	UIS.InputEnded:Connect(function(input)
		if not miniDragging then
			return
		end

		if input == miniDragInput
			or input.UserInputType == Enum.UserInputType.MouseButton1 then

			miniDragging = false
			miniDragInput = nil
			miniDragStart = nil
			miniStartAbsolute = nil
			miniStartUDimPosition = nil

			if miniWasDragged then
				miniBlockClickUntil = os.clock() + .20
			end
		end
	end)
end

local function minimizePanel()
	if menuDeleted or not panel.Visible then
		return
	end

		TweenService:Create(
			panelScale,
			TweenInfo.new(.12,Enum.EasingStyle.Quint,Enum.EasingDirection.In),
			{Scale = .92}
		):Play()
	end

	task.delay(.11,function()
		if menuDeleted then
			return
		end

		panel.Visible = false
		if panelScale then
			panelScale.Scale = 1
		end

		local miniScale = openButton:FindFirstChild("SmoothScale")
		if miniScale then
			miniScale.Scale = .72
		end

		openButton.Visible = true

				miniScale,
				TweenInfo.new(.18,Enum.EasingStyle.Back,Enum.EasingDirection.Out),
				{Scale = 1}
			):Play()
		end
	end)

	flightControls.Visible = flying
end

local function updateResize(inputPosition)
	if resizeLocked
		or not resizing
		or not resizeDirection
		or not resizeStart
		or not resizeStartPosition
		or not resizeStartUDimPosition
		or not resizeStartSize then
		return
	end
	local delta = inputPosition - resizeStart
	camera = workspace.CurrentCamera
	local viewport =
		camera and camera.ViewportSize
		or Vector2.new(1000,700)
	local startLeft = resizeStartPosition.X
	local startTop = resizeStartPosition.Y
	local startWidth = resizeStartSize.X
	local startHeight = resizeStartSize.Y
	local fromLeft =
		resizeDirection == "left"
		or resizeDirection == "topLeft"
		or resizeDirection == "bottomLeft"
	local fromRight =
		resizeDirection == "right"
		or resizeDirection == "topRight"
		or resizeDirection == "bottomRight"
	local fromTop =
		resizeDirection == "top"
		or resizeDirection == "topLeft"
		or resizeDirection == "topRight"
	local fromBottom =
		resizeDirection == "bottom"
		or resizeDirection == "bottomLeft"
		or resizeDirection == "bottomRight"
	local appliedX = 0
	local appliedY = 0
	local newWidth = startWidth
	local newHeight = startHeight
	if fromLeft then
		appliedX = math.clamp(
			delta.X,
			SCREEN_MARGIN - startLeft,
			startWidth - MIN_PANEL_WIDTH
		)
		newWidth = startWidth - appliedX
	elseif fromRight then
		newWidth = math.clamp(
			startWidth + delta.X,
			MIN_PANEL_WIDTH,
			viewport.X - startLeft - SCREEN_MARGIN
		)
	end
	if fromTop then
		appliedY = math.clamp(
			delta.Y,
			SCREEN_MARGIN - startTop,
			startHeight - MIN_PANEL_HEIGHT
		)
		newHeight = startHeight - appliedY
	elseif fromBottom then
		newHeight = math.clamp(
			startHeight + delta.Y,
			MIN_PANEL_HEIGHT,
			viewport.Y - startTop - SCREEN_MARGIN
		)
	end
	panel.Position = UDim2.new(
		resizeStartUDimPosition.X.Scale,
		resizeStartUDimPosition.X.Offset + appliedX,
		resizeStartUDimPosition.Y.Scale,
		resizeStartUDimPosition.Y.Offset + appliedY
	)
	panel.Size = UDim2.fromOffset(newWidth,newHeight)
end
UIS.InputChanged:Connect(function(input)
	if not resizing then
		return
	end
	if input.UserInputType == Enum.UserInputType.MouseMovement then
		updateResize(input.Position)
	elseif input.UserInputType == Enum.UserInputType.Touch
		and input == resizeInput then
		updateResize(input.Position)
	end
end)
UIS.InputEnded:Connect(function(input)
	if input == resizeInput
		or (
			resizing
			and input.UserInputType == Enum.UserInputType.MouseButton1
		) then
		resizing = false
		resizeInput = nil
		resizeDirection = nil
	end
end)
end

local function setupDragging()
local dragging = false
local dragInput
local dragStart

header.InputBegan:Connect(function(input)
	if movementLocked then
		return
	end
	local position = input.Position
	local closePos = closeButton.AbsolutePosition
	local closeSize = closeButton.AbsoluteSize
	local minPos = minimizeButton.AbsolutePosition
	local minSize = minimizeButton.AbsoluteSize
	local insideClose =
		position.X >= closePos.X
		and position.X <= closePos.X + closeSize.X
		and position.Y >= closePos.Y
		and position.Y <= closePos.Y + closeSize.Y
	local insideMin =
		position.X >= minPos.X
		and position.X <= minPos.X + minSize.X
		and position.Y >= minPos.Y
		and position.Y <= minPos.Y + minSize.Y
	if insideClose or insideMin then
		return
	end
	if input.UserInputType == Enum.UserInputType.Touch
		or input.UserInputType == Enum.UserInputType.MouseButton1 then
		dragging = true
		dragInput = input
		dragStart = input.Position
		dragStartPosition = panel.Position
	end
end)

	if movementLocked or not dragging then
		return
	end
	if input.UserInputType == Enum.UserInputType.MouseMovement
		or (
			input.UserInputType == Enum.UserInputType.Touch
			and input == dragInput

			dragStartPosition.X.Scale,
			dragStartPosition.X.Offset + delta.X,
			dragStartPosition.Y.Scale,
			dragStartPosition.Y.Offset + delta.Y
		)
	end
end)
UIS.InputEnded:Connect(function(input)
	if input == dragInput
		or input.UserInputType == Enum.UserInputType.MouseButton1 then
		dragging = false
		dragInput = nil
	end

		for _,part in ipairs(otherCharacter:GetDescendants()) do
			if part:IsA("BasePart") then
				if antiFlingOriginalCollisions[part] == nil then

		end

		antiFlingEnabled = false
		antiFlingToken += 1

			setToggle(antiFlingButton,false,"ANTI FLING")

	local function startAntiFling()
		antiFlingEnabled = true
		antiFlingToken += 1
		local myToken = antiFlingToken
		setToggle(antiFlingButton,true,"ANTI FLING")
		setStatus("ANTI FLING ON")

		task.spawn(function()
			while antiFlingEnabled
				and myToken == antiFlingToken
				and not menuDeleted do

				local currentCharacter = player.Character
				local currentHumanoid =
					currentCharacter
					and currentCharacter:FindFirstChildOfClass("Humanoid")
				local currentRoot =
					currentCharacter
					and currentCharacter:FindFirstChild("HumanoidRootPart")

				if currentRoot then
					local ownLinear =
						currentRoot.AssemblyLinearVelocity.Magnitude
					local ownAngular =
						currentRoot.AssemblyAngularVelocity.Magnitude

					if ownLinear < 70
						and ownAngular < 16
						and currentHumanoid
						and currentHumanoid.FloorMaterial
							~= Enum.Material.Air then
						antiFlingLastSafeCFrame = currentRoot.CFrame
					end

					-- If our character suddenly receives fling-level physics,
					-- kill the velocity and return to the last stable position.
					if ownLinear > 175 or ownAngular > 48 then
						currentRoot.AssemblyLinearVelocity = Vector3.zero
						currentRoot.AssemblyAngularVelocity = Vector3.zero

						if antiFlingLastSafeCFrame then
							currentRoot.CFrame = antiFlingLastSafeCFrame
						end

						setStatus("ANTI FLING // BLOCKED PHYSICS SPIKE")
					end

					for _,otherPlayer in ipairs(Players:GetPlayers()) do
						if otherPlayer ~= player then
							local otherCharacter = otherPlayer.Character
							local otherRoot =
								otherCharacter
								and otherCharacter:FindFirstChild("HumanoidRootPart")

							if otherRoot and otherCharacter then
								local distance =
									(otherRoot.Position-currentRoot.Position).Magnitude

								if distance <= 55 then
									local otherLinear =
										otherRoot.AssemblyLinearVelocity.Magnitude
									local otherAngular =
										otherRoot.AssemblyAngularVelocity.Magnitude

									if otherLinear > 110
										or otherAngular > 24 then
										suppressFlingCharacter(otherCharacter)
									end
								end
							end
						end
					end
				end

				task.wait(.06)
			end

			if myToken == antiFlingToken then
				restoreAntiFlingCollisions()
			end
		end)
	end

	local function startFakeLag()
		stopFakeLag()

		-- Fake Lag and the permanent Freeze button both use Anchored,
		-- so turn Freeze off first to avoid the two systems fighting.
		if frozenRoot and frozenRoot.Parent then
			frozenRoot.Anchored = false
			frozenRoot = nil
		end
		frozen = false
		setToggle(freezeButton,false,"FREEZE")

		fakeLagEnabled = true
		fakeLagToken += 1
		local token = fakeLagToken

		task.spawn(function()
			while fakeLagEnabled and fakeLagToken == token do
				if not root or not root.Parent then
					task.wait(.1)
					continue
				end

				local currentRoot = root
				fakeLagHeldRoot = currentRoot
				currentRoot.Anchored = true

				local elapsed = 0
				while fakeLagEnabled
					and fakeLagToken == token
					and currentRoot.Parent
					and elapsed < fakeLagHold do

					local step = math.min(.05,fakeLagHold - elapsed)
					task.wait(step)
					elapsed += step
				end

				if currentRoot.Parent and currentRoot.Anchored and not frozen then
					currentRoot.Anchored = false
				end
				fakeLagHeldRoot = nil

				if fakeLagEnabled and fakeLagToken == token then
					task.wait(fakeLagRelease)
				end
			end

			releaseFakeLagRoot()
		end)
	end

			or object:IsA("Sparkles")

		if not performanceMode or not isPerformanceObject(object) then
			return
		end
		if performanceStates[object] == nil then
			performanceStates[object] = object.Enabled
		end
		object.Enabled = false
	end

		for object,enabled in pairs(performanceStates) do
			if object and object.Parent then
				object.Enabled = enabled
			end

		table.clear(performanceStates)
		Lighting.GlobalShadows = originalShadows
	end

local function setupPowerSuite()
	local HttpService = game:GetService("HttpService")

	local state = {
		file = "not_ruby_v4_power_"..tostring(player.UserId)..".json",
		canFiles =
			type(isfile) == "function"
			and type(readfile) == "function"
			and type(writefile) == "function",
		data = {
			favorites = {},
			keybinds = {
				Flight = "F",
				Noclip = "N",
				Invisible = "X",
				FakeLag = "L",
				WallParts = "V"
			},
			waypoints = {},
			presets = {},
			settings = {}
		},
		registry = {},
		waypointObjects = {},
		stats = {
			started = os.clock(),

			minFps = 999,
			lastPosition = nil
		}
	}

	local ui = {}

	-- --------------------------------------------------------
	-- Storage
	-- --------------------------------------------------------
	local function loadStorage()
		if not state.canFiles then
			return
		end

		pcall(function()
			if not isfile(state.file) then
				return
			end

			local decoded =
				HttpService:JSONDecode(readfile(state.file))

			if typeof(decoded) ~= "table" then
				return
			end

			for key,value in pairs(decoded) do
				state.data[key] = value
			end

			state.data.favorites =
				typeof(state.data.favorites) == "table"
				and state.data.favorites
				or {}
			state.data.keybinds =
				typeof(state.data.keybinds) == "table"
				and state.data.keybinds
				or {}
			state.data.waypoints =
				typeof(state.data.waypoints) == "table"
				and state.data.waypoints
				or {}
			state.data.presets =
				typeof(state.data.presets) == "table"
				and state.data.presets
				or {}
			state.data.settings =
				typeof(state.data.settings) == "table"
				and state.data.settings
				or {}

			local defaults = {
				Flight = "F",
				Noclip = "N",
				Invisible = "X",
				FakeLag = "L",
				WallParts = "V"
			}

			for feature,keyName in pairs(defaults) do
				if type(state.data.keybinds[feature]) ~= "string" then
					state.data.keybinds[feature] = keyName
				end
			end
		end)
	end

	local function saveStorage()
		if not state.canFiles then
			return false
		end

		local ok = pcall(function()
			writefile(
				state.file,
				HttpService:JSONEncode(state.data)
			)
		end)

		return ok
	end

	loadStorage()

	-- --------------------------------------------------------

		end

		local toast = Instance.new("Frame")

	local function cleanLabel(value)
		value = tostring(value or "")
		value = string.gsub(value,"%s+%[ON%]","")
		value = string.gsub(value,"%s+%[OFF%]","")
		value = string.gsub(value,"%s+"," ")
		return value
	end

	local function pageForObject(object)
		for pageName,page in pairs(pages) do
			if object:IsDescendantOf(page) then
				return pageName,page
			end
		end
		return nil,nil
	end

	local function jumpToFeature(entry)
		if not entry
			or not entry.button
			or not entry.button.Parent then
			return
		end

		local page = pages[entry.page]
		if not page then
			return
		end

		setTab(entry.page)

		task.defer(function()
			task.wait()
			local relative =
				entry.button.AbsolutePosition.Y
				- page.AbsolutePosition.Y
				+ page.CanvasPosition.Y
				- 55

			page.CanvasPosition =
				Vector2.new(0,math.max(0,relative))

			local scale =
				entry.button:FindFirstChildOfClass("UIScale")

			if scale then
				scale.Scale = .92
				TweenService:Create(
					scale,
					TweenInfo.new(
						.22,
						Enum.EasingStyle.Back,
						Enum.EasingDirection.Out
					),
					{Scale = 1}
				):Play()
			end
		end)
	end

				and not child:IsA("UIPadding") then

			end
		end
	end

	local function makeFavoriteStar(button,pageName)
		if button:GetAttribute("NotRubyFavoriteStarred") then
			return
		end

		if button.Name == "FavoriteStar"
			or button.Text == ""
			or button:IsDescendantOf(ui.favoritesPage)
			or button:IsDescendantOf(ui.waypointsPage)
			or button:IsDescendantOf(ui.statsPage) then
			return
		end

		local label = cleanLabel(button.Text)
		if label == "" then
			return
		end

		local key = pageName.."|"..label

		button:SetAttribute("NotRubyFavoriteStarred",true)
		button:SetAttribute("NotRubyFeatureKey",key)

		state.registry[key] = {
			button = button,
			page = pageName,
			label = label
		}

		local star = Instance.new("TextButton")
		star.Name = "FavoriteStar"
		star.AnchorPoint = Vector2.new(1,0)
		star.Position = UDim2.new(1,-3,0,3)
		star.Size = UDim2.fromOffset(21,21)
		star.Text =
			state.data.favorites[key]
			and "★"
			or "☆"
		star.TextSize = 13
		star.Font = Enum.Font.GothamBlack
		star.TextColor3 = accent
		star.BackgroundColor3 = PANEL
		star.BackgroundTransparency = .14
		star.BorderSizePixel = 0
		star.AutoButtonColor = false
		star.ZIndex = button.ZIndex + 10
		star.Parent = button
		corner(star,7)

		star.Activated:Connect(function()
			if state.data.favorites[key] then
				state.data.favorites[key] = nil
				star.Text = "☆"
				notify("REMOVED FROM FAVORITES // "..label)
			else

				notify("ADDED TO FAVORITES // "..label)
			end

			saveStorage()
			if rebuildFavorites then

			end
		end)
	end

			end
		end
	end

	local function rebuildSearch()
		clearChildren(ui.searchResults)

		local query =
			string.lower(
				string.gsub(
					ui.searchBox.Text or "",
					"^%s*(.-)%s*$",
					"%1"
				)
			)

		if query == "" then
			ui.searchResults.Visible = false
			ui.searchResults.Size = UDim2.new(1,-147,0,0)
			return
		end

		local matches = {}

		for _,entry in pairs(state.registry) do
			local haystack =
				string.lower(
					entry.label.." "..entry.page
				)

			if string.find(haystack,query,1,true) then
				table.insert(matches,entry)
			end
		end

		table.sort(matches,function(a,b)
			return a.label < b.label
		end)

		local shown = math.min(#matches,6)

		ui.searchResults.Visible = true
		ui.searchResults.Size =
			UDim2.new(
				1,
				-147,
				0,
				shown > 0 and (shown*41 + 10) or 54
			)

		for index,entry in ipairs(matches) do
			if index > 24 then
				break
			end

			local result = Instance.new("TextButton")
			result.Size = UDim2.new(1,-10,0,36)
			result.Text =
				"["..entry.page.."]  "..entry.label
			result.TextSize = 9
			result.Font = Enum.Font.GothamBold
			result.TextColor3 = TEXT
			result.TextXAlignment =
				Enum.TextXAlignment.Left
			result.BackgroundColor3 = SURFACE_2
			result.BorderSizePixel = 0
			result.AutoButtonColor = false
			result.LayoutOrder = index
			result.ZIndex = 251
			result.Parent = ui.searchResults
			corner(result,9)

			local padding = Instance.new("UIPadding")
			padding.PaddingLeft = UDim.new(0,11)
			padding.PaddingRight = UDim.new(0,8)
			padding.Parent = result

			result.Activated:Connect(function()
				ui.searchBox.Text = ""
				ui.searchResults.Visible = false
				jumpToFeature(entry)
			end)
		end

		if #matches == 0 then
			local empty = Instance.new("TextLabel")
			empty.Size = UDim2.new(1,-10,0,40)
			empty.BackgroundTransparency = 1
			empty.Text = "NO FEATURE FOUND"
			empty.TextSize = 9
			empty.Font = Enum.Font.GothamBold
			empty.TextColor3 = MUTED
			empty.ZIndex = 251
			empty.Parent = ui.searchResults
		end
	end

		fpsFrames += 1

		local now = os.clock()
		local elapsed = now - fpsStarted

		if elapsed >= .5 then
			state.stats.fps =
				math.floor(fpsFrames / elapsed + .5)

			if state.stats.fps > 0 then
				state.stats.minFps =
					math.min(
						state.stats.minFps,
						state.stats.fps
					)
			end

			fpsFrames = 0
			fpsStarted = now
		end
	end)

			local currentRoot =

				and player.Character:FindFirstChild("HumanoidRootPart")

			if currentRoot then
				local position = currentRoot.Position
				local speed =
					Vector3.new(
						currentRoot.AssemblyLinearVelocity.X,
						0,
						currentRoot.AssemblyLinearVelocity.Z
					).Magnitude

				state.stats.maxSpeed =
					math.max(state.stats.maxSpeed,speed)

				if state.stats.lastPosition then
					local delta =
						(position-state.stats.lastPosition).Magnitude

					-- Ignore huge teleports when counting traveled distance.
					if delta < 120 then
						state.stats.distance += delta
					end
				end

				state.stats.lastPosition = position

				local seconds =
					math.floor(os.clock()-state.stats.started)
				local hours = math.floor(seconds/3600)
				local minutes =
					math.floor((seconds%3600)/60)
				local secs = seconds%60

				ui.statTime.Text =
					string.format(
						"%02d:%02d:%02d",
						hours,
						minutes,
						secs
					)
				ui.statDistance.Text =
					tostring(math.floor(state.stats.distance))
					.." studs"
				ui.statSpeed.Text =
					tostring(math.floor(speed+.5))
					.." / "

				ui.statJumps.Text =
					tostring(state.stats.jumps)
				ui.statFps.Text =
					tostring(state.stats.fps)

					..(

						or tostring(state.stats.minFps)
					)

		end
	end)

	local function applyLanguageChoice()
		local query = string.lower(
			string.gsub(languageInput.Text or "","^%s*(.-)%s*$","%1")
		)

		if query == "" then
			setStatus("TYPE A LANGUAGE OR COUNTRY")
			return
		end

		local choice = nil

		for _,entry in ipairs(languageChoices) do
			local haystack =
				string.lower(
					entry.code.." "..entry.name.." "..entry.country
				)

			if query == string.lower(entry.code)
				or query == string.lower(entry.name)
				or query == string.lower(entry.country)
				or string.find(haystack,query,1,true) then
				choice = entry
				break
			end
		end

		if not choice then
			setStatus("LANGUAGE NOT FOUND")
			return
		end

		gui:SetAttribute("WelcomeLanguageCode",choice.code)
		gui:SetAttribute("WelcomeLanguageName",choice.name)
		gui:SetAttribute("WelcomeCountry",choice.country)

		if type(writefile) == "function" then
			pcall(function()
				writefile(languageFile,choice.code)
			end)
		end

		languageInput.Text = ""
		languageCurrent.Text = "CURRENT: "..choice.name
		setStatus("LANGUAGE CHANGED // "..string.upper(choice.name))
	end

	local function triggerCommand(feature)
		if feature == "Flight" then
			setFlight(not flying)
		elseif feature == "Noclip" then
			setNoclip(not noclip)
		elseif feature == "Invisible" then
			gui:SetAttribute(
				"CommandInvisible",
				(tonumber(gui:GetAttribute("CommandInvisible")) or 0)+1
			)
		elseif feature == "FakeLag" then
			gui:SetAttribute(
				"CommandFakeLag",
				(tonumber(gui:GetAttribute("CommandFakeLag")) or 0)+1
			)
		elseif feature == "WallParts" then
			gui:SetAttribute(
				"WallPartsVisual",
				not (gui:GetAttribute("WallPartsVisual") == true)
			)
		end
	end

	local function colorToData(color)
		return {
			r = math.floor(color.R*255+.5),
			g = math.floor(color.G*255+.5),
			b = math.floor(color.B*255+.5)
		}
	end

	local function dataToColor(data,fallback)
		if typeof(data) ~= "table" then
			return fallback
		end

		return Color3.fromRGB(
			math.clamp(tonumber(data.r) or 0,0,255),
			math.clamp(tonumber(data.g) or 0,0,255),
			math.clamp(tonumber(data.b) or 0,0,255)
		)
	end

	local function captureSettings()
		local saved = {
			themeIndex = themeIndex,
			accent = colorToData(accent),
			outlineMode =
				gui:GetAttribute("OutlineMode"),
			outlineIndex =
				gui:GetAttribute("OutlineIndex"),
			outline =
				colorToData(
					gui:GetAttribute("OutlineColor")
					or accent
				),
			panel = {
				xs = panel.Position.X.Scale,
				xo = panel.Position.X.Offset,
				ys = panel.Position.Y.Scale,
				yo = panel.Position.Y.Offset,
				w = panel.AbsoluteSize.X,
				h = panel.AbsoluteSize.Y
			},
			mini = {
				xs = openButton.Position.X.Scale,
				xo = openButton.Position.X.Offset,
				ys = openButton.Position.Y.Scale,
				yo = openButton.Position.Y.Offset,
				w = openButton.AbsoluteSize.X,
				h = openButton.AbsoluteSize.Y
			}
		}

		local hud = gui:FindFirstChild("NotRubyLiveHUD")
		if hud and hud:IsA("GuiObject") then
			saved.hud = {
				xs = hud.Position.X.Scale,
				xo = hud.Position.X.Offset,
				ys = hud.Position.Y.Scale,
				yo = hud.Position.Y.Offset,
				w = hud.AbsoluteSize.X,
				h = hud.AbsoluteSize.Y
			}
		end

		state.data.settings = saved
	end

	local function applySavedSettings()
		local saved = state.data.settings
		if typeof(saved) ~= "table" then
			return
		end

		if saved.accent then
			accent =
				dataToColor(saved.accent,accent)

			local savedIndex =
				tonumber(saved.themeIndex)

			if savedIndex
				and savedIndex >= 1
				and savedIndex <= #themes then
				themeIndex = savedIndex
				themeButton.Text =
					"THEME: "
					..themes[themeIndex].name
			else
				themeIndex = 0
				themeButton.Text = "THEME: CUSTOM"
			end

			status.TextColor3 = accent
			sidebar.ScrollBarImageColor3 = accent
			applyTabVisuals()
			refreshToggleColors()
		end

		if saved.outline then
			local outline =
				dataToColor(saved.outline,accent)

			gui:SetAttribute(
				"OutlineMode",
				saved.outlineMode or "MATCH"
			)
			gui:SetAttribute(
				"OutlineIndex",
				tonumber(saved.outlineIndex) or 1
			)
			gui:SetAttribute("OutlineColor",outline)

			for _,object in ipairs(gui:GetDescendants()) do
				if object:IsA("UIStroke") then
					object.Color = outline
				end
			end

			outlineButton.Text =
				saved.outlineMode == "CUSTOM"
				and "OUTLINE: CUSTOM"
				or "OUTLINE: MATCH"
		end

		if typeof(saved.panel) == "table" then
			panel.Position =
				UDim2.new(
					tonumber(saved.panel.xs) or .5,
					tonumber(saved.panel.xo) or -280,
					tonumber(saved.panel.ys) or .5,
					tonumber(saved.panel.yo) or -215
				)
			panel.Size =
				UDim2.fromOffset(
					math.max(
						320,
						tonumber(saved.panel.w) or 560
					),
					math.max(
						250,
						tonumber(saved.panel.h) or 430
					)
				)
		end

		if typeof(saved.mini) == "table" then
			openButton.Position =
				UDim2.new(
					tonumber(saved.mini.xs) or 0,
					tonumber(saved.mini.xo) or 9,
					tonumber(saved.mini.ys) or .5,
					tonumber(saved.mini.yo) or -26
				)

			local size =
				math.clamp(
					tonumber(saved.mini.w) or 52,
					34,
					140
				)

			openButton.Size =
				UDim2.fromOffset(size,size)
			openButton.TextSize =
				math.clamp(
					math.floor(size*.46),
					16,
					42
				)
		end

		local hud = gui:FindFirstChild("NotRubyLiveHUD")
		if hud
			and hud:IsA("GuiObject")
			and typeof(saved.hud) == "table" then

			hud.Position =
				UDim2.new(
					tonumber(saved.hud.xs) or 0,
					tonumber(saved.hud.xo) or 18,
					tonumber(saved.hud.ys) or 0,
					tonumber(saved.hud.yo) or 82
				)
			hud.Size =
				UDim2.fromOffset(
					math.max(
						210,
						tonumber(saved.hud.w) or 300
					),
					math.max(
						100,
						tonumber(saved.hud.h) or 128
					)
				)
		end
	end

end

	-- Flags are generated with Regional Indicator Unicode characters.
	-- This avoids hard-coding flag emoji bytes into the source.
	local function countryFlag(countryCode)
		countryCode = string.upper(tostring(countryCode or ""))
		if #countryCode ~= 2 then
			return ""
		end

		local first = string.byte(countryCode,1)
		local second = string.byte(countryCode,2)

		if not first or not second
			or first < 65 or first > 90
			or second < 65 or second > 90 then
			return ""
		end

		return utf8.char(
			0x1F1E6 + first - 65,
			0x1F1E6 + second - 65
		)
	end

	local localeAliases = {
		nb = "no",
		nn = "no"
	}

	local function findLanguageIndex(code)
		code = string.lower(tostring(code or "en"))
		code = string.match(code,"^[a-z]+") or "en"
		code = localeAliases[code] or code

		for index,language in ipairs(languages) do
			if language.code == code then
				return index
			end
		end

		return 1
	end

	local function normalizeSearch(value)
		value = string.lower(tostring(value or ""))
		value = string.gsub(value,"%s+"," ")
		value = string.gsub(value,"^%s+","")
		value = string.gsub(value,"%s+$","")
		return value
	end

	local function clearSearchResults()
		for _,child in ipairs(searchResults:GetChildren()) do
			if child:IsA("GuiButton") or child.Name == "NoCountryResult" then
				child:Destroy()
			end
		end
	end

	local function updateLanguage()
		local language = languages[languageIndex]
		if not language then
			return
		end

		local flag = countryFlag(language.countryCode)

		if visitCount > 1 then
			welcomeText.Text =
				"WELCOME BACK // VISIT #"..tostring(visitCount)
		else
			welcomeText.Text = language.welcome
		end
		welcomeSubtitle.Text = language.subtitle
		languageCaption.Text = language.language
		countrySearch.PlaceholderText = language.search

		languageButton.Text =
			flag
			.."  "
			..language.name
			.."  •  "
			..language.country
			.."  ("
			..tostring(languageIndex)
			.."/"
			..tostring(#languages)
			..")"

		enterButton.Text = language.enter

		gui:SetAttribute("WelcomeLanguageCode",language.code)
		gui:SetAttribute("WelcomeLanguageName",language.name)
		gui:SetAttribute("WelcomeCountry",language.country)
		gui:SetAttribute("WelcomeCountryCode",language.countryCode)

		if type(writefile) == "function" then
			pcall(function()
				writefile(languageFile,language.code)
			end)
		end
	end

	local function selectLanguage(index,clearSearch)
		if not languages[index] then
			return
		end

		languageIndex = index
		updateLanguage()

		if clearSearch then
			countrySearch.Text = ""
			searchResults.Visible = false
			clearSearchResults()
		end

		local scale = languageButton:FindFirstChildOfClass("UIScale")
		if not scale then
			scale = Instance.new("UIScale")
			scale.Parent = languageButton
		end

		scale.Scale = .97
		TweenService:Create(
			scale,
			TweenInfo.new(.14,Enum.EasingStyle.Back,Enum.EasingDirection.Out),
			{Scale = 1}
		):Play()
	end

	local function changeLanguage(direction)
		local newIndex = languageIndex + direction

		if newIndex < 1 then
			newIndex = #languages
		elseif newIndex > #languages then
			newIndex = 1
		end

		selectLanguage(newIndex,true)
	end

	local function rebuildCountrySearch()
		searchSerial += 1
		local mySerial = searchSerial
		local query = normalizeSearch(countrySearch.Text)

		clearSearchResults()

		if query == "" then
			searchResults.Visible = false
			return
		end

		local matches = {}

		for index,language in ipairs(languages) do
			local haystack = normalizeSearch(
				language.country
				.." "
				..language.name
				.." "
				..language.code
				.." "
				..language.countryCode
				.." "
				..tostring(language.aliases or "")
			)

			if string.find(haystack,query,1,true) then
				table.insert(matches,index)
			end
		end

		if mySerial ~= searchSerial then
			return
		end

		searchResults.Visible = true

		if #matches == 0 then
			local empty = Instance.new("TextLabel")
			empty.Name = "NoCountryResult"
			empty.Size = UDim2.new(1,-10,0,34)
			empty.BackgroundTransparency = 1
			empty.Text = "NO COUNTRY FOUND"
			empty.TextSize = 9
			empty.Font = Enum.Font.GothamBold
			empty.TextColor3 = MUTED
			empty.ZIndex = 521
			empty.Parent = searchResults
			return
		end

		for order,index in ipairs(matches) do
			local language = languages[index]
			local result = Instance.new("TextButton")
			result.Name = "CountryResult_"..tostring(index)
			result.Size = UDim2.new(1,-10,0,36)
			result.BackgroundColor3 = SURFACE_2
			result.BorderSizePixel = 0
			result.AutoButtonColor = false
			result.Text =
				countryFlag(language.countryCode)
				.."  "
				..language.country
				.."  •  "
				..language.name
			result.TextSize = 9
			result.Font = Enum.Font.GothamBold
			result.TextColor3 = TEXT
			result.TextXAlignment = Enum.TextXAlignment.Left
			result.LayoutOrder = order
			result.ZIndex = 521
			result.Parent = searchResults
			corner(result,8)

			local resultPadding = Instance.new("UIPadding")
			resultPadding.PaddingLeft = UDim.new(0,10)
			resultPadding.PaddingRight = UDim.new(0,8)
			resultPadding.Parent = result

			result.Activated:Connect(function()
				selectLanguage(index,true)
			end)
		end
	end

	-- Auto-select a supported language from the player's Roblox locale.
	local localeOk,localeId = pcall(function()

	local locatorFolder = Instance.new("Folder")
	locatorFolder.Name = "NotRubyShooterLocators"
	locatorFolder.Parent = workspace

	local locatorObjects = {}

	local function isSameTeam(otherPlayer)
		if otherPlayer == player then
			return true
		end

		if player.Neutral or otherPlayer.Neutral then
			return false
		end

		return player.Team ~= nil
			and otherPlayer.Team ~= nil
			and player.Team == otherPlayer.Team
	end

	local function clearLocator(targetPlayer)
		local bundle = locatorObjects[targetPlayer]
		if not bundle then
			return
		end

		for _,object in pairs(bundle) do
			if typeof(object) == "Instance"
				and object.Parent then
				object:Destroy()
			end
		end

		locatorObjects[targetPlayer] = nil
	end

	local function clearAllLocators()
		for targetPlayer in pairs(locatorObjects) do
			clearLocator(targetPlayer)
		end
	end

	local function shouldShowPlayer(otherPlayer)
		if otherPlayer == player then
			return false,nil
		end

		local sameTeam = isSameTeam(otherPlayer)

		if sameTeam and state.teamLocator then
			return true,Color3.fromRGB(65,225,120)
		end

		if not sameTeam and state.enemyLocator then
			return true,Color3.fromRGB(255,55,70)
		end

		return false,nil
	end

	local function createLocator(otherPlayer,color)
		clearLocator(otherPlayer)

		local character = otherPlayer.Character
		if not character then
			return
		end

		local humanoid =
			character:FindFirstChildOfClass("Humanoid")
		local head = character:FindFirstChild("Head")
		local root =
			character:FindFirstChild("HumanoidRootPart")

		if not humanoid or not root then
			return
		end

		local highlight = Instance.new("Highlight")
		highlight.Name =
			"NotRubyShooterHighlight_"
			..tostring(otherPlayer.UserId)
		highlight.Adornee = character
		highlight.FillColor = color
		highlight.OutlineColor = color
		highlight.FillTransparency = .78
		highlight.OutlineTransparency = .05
		highlight.DepthMode =
			Enum.HighlightDepthMode.AlwaysOnTop
		highlight.Parent = locatorFolder

		local billboard = Instance.new("BillboardGui")
		billboard.Name =
			"NotRubyShooterLabel_"
			..tostring(otherPlayer.UserId)
		billboard.Adornee = head or root
		billboard.Size = UDim2.fromOffset(220,42)
		billboard.StudsOffset = Vector3.new(0,2.7,0)
		billboard.AlwaysOnTop = true
		billboard.Parent = locatorFolder

		local label = Instance.new("TextLabel")
		label.Name = "Label"
		label.Size = UDim2.fromScale(1,1)
		label.BackgroundTransparency = 1
		label.Text = otherPlayer.DisplayName
		label.TextSize = 10
		label.Font = Enum.Font.GothamBlack
		label.TextColor3 = color
		label.TextStrokeTransparency = .28
		label.Parent = billboard

		locatorObjects[otherPlayer] = {
			highlight = highlight,
			billboard = billboard,
			label = label,
			root = root
		}
	end

	local function refreshLocators()
		for _,otherPlayer in ipairs(Players:GetPlayers()) do
			local show,color =
				shouldShowPlayer(otherPlayer)

			if show then
				local bundle = locatorObjects[otherPlayer]
				local character = otherPlayer.Character

				if not bundle
					or not bundle.highlight
					or not bundle.highlight.Parent
					or bundle.highlight.Adornee ~= character then
					createLocator(otherPlayer,color)
				else
					bundle.highlight.FillColor = color
					bundle.highlight.OutlineColor = color
					bundle.label.TextColor3 = color
				end
			else
				clearLocator(otherPlayer)
			end
		end

		for targetPlayer in pairs(locatorObjects) do
			if targetPlayer.Parent ~= Players then
				clearLocator(targetPlayer)
			end
		end
	end

	local function isAimTargetVisible(targetPart)
		camera = workspace.CurrentCamera
		if not camera or not targetPart then
			return false
		end

		local origin = camera.CFrame.Position
		local direction = targetPart.Position - origin

		local params = RaycastParams.new()
		params.FilterType = Enum.RaycastFilterType.Exclude
		params.FilterDescendantsInstances = {
			player.Character
		}

		local result = workspace:Raycast(origin,direction,params)

		if not result then
			return true
		end

		return result.Instance
			and result.Instance:IsDescendantOf(targetPart.Parent)
	end

	local function getBestAimTarget()
		camera = workspace.CurrentCamera
		if not camera then
			return nil
		end

		local viewport = camera.ViewportSize
		local center =
			Vector2.new(viewport.X*.5,viewport.Y*.5)

		local bestPart = nil
		local bestDistance = state.fovRadius

		for _,otherPlayer in ipairs(Players:GetPlayers()) do
			if otherPlayer ~= player
				and not isSameTeam(otherPlayer) then

				local character = otherPlayer.Character
				local humanoid =
					character
					and character:FindFirstChildOfClass("Humanoid")

				local targetPart =
					character
					and (
						character:FindFirstChild(state.aimTarget)
						or character:FindFirstChild("HumanoidRootPart")
					)

				if humanoid
					and humanoid.Health > 0
					and targetPart
					and targetPart:IsA("BasePart") then

					local screenPosition,onScreen =
						camera:WorldToViewportPoint(targetPart.Position)

					if onScreen and screenPosition.Z > 0 then
						local screenDistance =
							(Vector2.new(
								screenPosition.X,
								screenPosition.Y
							)-center).Magnitude

						if screenDistance <= bestDistance
							and isAimTargetVisible(targetPart) then
							bestDistance = screenDistance
							bestPart = targetPart
						end
					end
				end
			end
		end

		return bestPart
	end

	local function getAimbotTargetPart()
		camera = workspace.CurrentCamera
		if not camera then
			state.lockedTarget = nil
			return nil
		end

		local viewport = camera.ViewportSize
		local center =
			Vector2.new(viewport.X*.5,viewport.Y*.5)

		local function validTarget(otherPlayer)
			if not otherPlayer
				or otherPlayer == player
				or otherPlayer.Parent ~= Players
				or isSameTeam(otherPlayer) then
				return nil
			end

			local character = otherPlayer.Character
			local humanoid =
				character
				and character:FindFirstChildOfClass("Humanoid")
			local targetPart =
				character
				and (
					character:FindFirstChild(state.aimTarget)
					or character:FindFirstChild("HumanoidRootPart")
				)

			if not humanoid
				or humanoid.Health <= 0
				or not targetPart
				or not targetPart:IsA("BasePart") then
				return nil
			end

			local screenPosition,onScreen =
				camera:WorldToViewportPoint(targetPart.Position)

			if not onScreen or screenPosition.Z <= 0 then
				return nil
			end

			local distance =
				(Vector2.new(
					screenPosition.X,
					screenPosition.Y
				)-center).Magnitude

			if distance > state.fovRadius then
				return nil
			end

			if not isAimTargetVisible(targetPart) then
				return nil
			end

			return targetPart
		end

		local currentPart = validTarget(state.lockedTarget)
		if currentPart then
			return currentPart
		end

		state.lockedTarget = nil

		local bestPlayer = nil
		local bestPart = nil
		local bestDistance = state.fovRadius

		for _,otherPlayer in ipairs(Players:GetPlayers()) do
			local part = validTarget(otherPlayer)

			if part then
				local screenPosition =
					camera:WorldToViewportPoint(part.Position)

				local distance =
					(Vector2.new(
						screenPosition.X,
						screenPosition.Y
					)-center).Magnitude

				if distance < bestDistance then
					bestDistance = distance
					bestPlayer = otherPlayer
					bestPart = part
				end
			end
		end

		state.lockedTarget = bestPlayer
		return bestPart
	end

	local function setAimbot(value)
		if not aimAssistAllowed then
			state.aimbot = false
			state.lockedTarget = nil
			aimbotButton.Text = "AIMBOT [LOCKED]"
			setStatus(
				"SET AIM_ASSIST_PLACE_ID TO YOUR GAME PLACEID"
			)
			return
		end

		state.aimbot = value == true
		state.lockedTarget = nil

		if state.aimbot and state.aimAssist then
			state.aimAssist = false
			setToggle(
				aimAssistButton,
				false,
				"AIM ASSIST"
			)
		end

		setToggle(
			aimbotButton,
			state.aimbot,
			"AIMBOT"
		)

		setStatus(
			state.aimbot
			and "AIMBOT ON // OWN-GAME PLACEID VERIFIED"
			or "AIMBOT OFF"
		)
	end

	local function setAimAssist(value)
		if not aimAssistAllowed then
			state.aimAssist = false
			aimAssistButton.Text = "AIM ASSIST [LOCKED]"
			setStatus(
				"SET AIM_ASSIST_PLACE_ID TO YOUR GAME PLACEID"
			)
			return
		end

		state.aimAssist = value == true

		if state.aimAssist and state.aimbot then
			state.aimbot = false
			state.lockedTarget = nil
			setToggle(
				aimbotButton,
				false,
				"AIMBOT"
			)
		end

		setToggle(
			aimAssistButton,
			state.aimAssist,
			"AIM ASSIST"
		)

		setStatus(
			state.aimAssist
			and "AIM ASSIST ON // OWN-GAME PLACEID VERIFIED"
			or "AIM ASSIST OFF"
		)
	end

			return

			refreshLocators

		otherPlayer:GetPropertyChangedSignal("Neutral"):Connect(
			refreshLocators
		)

		refreshLocators()
	end)

		clearLocator(otherPlayer)
	end)

	player:GetPropertyChangedSignal("Neutral"):Connect(refreshLocators)

	for _,otherPlayer in ipairs(Players:GetPlayers()) do
		if otherPlayer ~= player then
			otherPlayer.CharacterAdded:Connect(function()
				task.wait(.2)
				refreshLocators()
			end)

			otherPlayer:GetPropertyChangedSignal("Team"):Connect(
				refreshLocators
			)
			otherPlayer:GetPropertyChangedSignal("Neutral"):Connect(
				refreshLocators
			)
		end
	end

	task.spawn(function()
		while gui.Parent and not menuDeleted do
			local localRoot =
				player.Character
				and player.Character:FindFirstChild("HumanoidRootPart")

			if state.teamLocator or state.enemyLocator then
				refreshLocators()

				if localRoot then
					for _,bundle in pairs(locatorObjects) do
						if bundle.root
							and bundle.root.Parent
							and bundle.label
							and bundle.label.Parent then

							local distance =
								math.floor(
									(localRoot.Position-bundle.root.Position)
									.Magnitude
								)

							local targetPlayer = nil
							for candidate,candidateBundle in pairs(locatorObjects) do
								if candidateBundle == bundle then
									targetPlayer = candidate
									break
								end
							end

							if targetPlayer then
								bundle.label.Text =
									targetPlayer.DisplayName
									.." // "
									..tostring(distance)
									.." studs"
							end
						end
					end
				end
			elseif next(locatorObjects) ~= nil then
				clearAllLocators()
			end

			local outline =
				gui:GetAttribute("OutlineColor") or accent
			fovStroke.Color = outline

			task.wait(.35)
		end
	end)

	RunService.RenderStepped:Connect(function()
		if not aimAssistAllowed or menuDeleted then
			return
		end

		camera = workspace.CurrentCamera
		if not camera then
			return
		end

		if state.aimbot then
			local targetPart = getAimbotTargetPart()

			if targetPart then
				local origin = camera.CFrame.Position
				camera.CFrame =
					CFrame.lookAt(
						origin,
						targetPart.Position
					)
			end

			return
		end

		if not state.aimAssist then
			return
		end

		local targetPart = getBestAimTarget()
		if not targetPart then
			return
		end

		local origin = camera.CFrame.Position
		local desired =
			CFrame.lookAt(origin,targetPart.Position)

		camera.CFrame =
			camera.CFrame:Lerp(
				desired,
				math.clamp(state.aimStrength,0,1)
			)
	end)

		aimAssistButton.BackgroundColor3 = SURFACE_2

		aimbotButton.BackgroundColor3 = SURFACE_2
		aimbotButton.TextColor3 = MUTED
	end

		state.aimbot = false
		state.aimAssist = false
		state.lockedTarget = nil
		clearAllLocators()
		if locatorFolder and locatorFolder.Parent then
			locatorFolder:Destroy()
		end
	end)
